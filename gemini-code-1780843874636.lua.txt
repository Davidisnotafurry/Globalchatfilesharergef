-- [[ CONFIGURATION FILE SYSTEM ENGINE ]] --
local function serializeConfig()
    local blockedList = {}
    for user, _ in pairs(shared.IgnoredPlayers) do
        table.insert(blockedList, string.format('"%s"', user))
    end
    local blockedStr = "{" .. table.concat(blockedList, ",") .. "}"
    
    local lines = {
        "BlockedUsers: " .. blockedStr,
        "JoinEnabled: " .. tostring(shared.AllowJoiningToggle),
        "Notifywhenplayerjoinsyourserver: " .. tostring(shared.Config.Notifywhenplayerjoinsyourserver),
        "Notifywhennewmessageonchat: " .. tostring(shared.Config.Notifywhennewmessageonchat),
        "Notifywhenplayerleavesyourserver: " .. tostring(shared.Config.Notifywhenplayerleavesyourserver),
        "Notifywhennewfileispublished: " .. tostring(shared.Config.Notifywhennewfileispublished),
        "Notificationsoundid: " .. tostring(shared.Config.Notificationsoundid),
        "Notificationsoundvolume: " .. tostring(shared.Config.Notificationsoundvolume),
        "Notificationsoundpitchmax: " .. tostring(shared.Config.Notificationsoundpitchmax),
        "Notificationsoundpitchmin: " .. tostring(shared.Config.Notificationsoundpitchmin)
    }
    return table.concat(lines, "\n")
end

local function saveConfig()
    pcall(function()
        writefile(SETTINGS_FILE, serializeConfig())
    end)
end

local function loadConfig()
    if not isfile(SETTINGS_FILE) then
        saveConfig()
        return
    end
    pcall(function()
        local data = readfile(SETTINGS_FILE)
        for line in string.gmatch(data, "[^\r\n]+") do
            local key, val = string.match(line, "([^:]+):%s*(.*)")
            if key and val then
                key = key:gsub("%s+", "")
                if key == "BlockedUsers" then
                    local listStr = string.match(val, "{(.*)}")
                    if listStr then
                        for user in string.gmatch(listStr, '"([^"]+)"') do
                            shared.IgnoredPlayers[user] = true
                        end
                    end
                elseif key == "JoinEnabled" then
                    shared.AllowJoiningToggle = (val == "true")
                elseif shared.Config[key] ~= nil then
                    if val == "true" or val == "false" then
                        shared.Config[key] = (val == "true")
                    else
                        shared.Config[key] = tonumber(val) or val
                    end
                end
            end
        end
    end)
end

loadConfig()

local function firebaseRequest(method, path, body)
    local url = FIREBASE_URL .. path .. ".json"
    local success, response = pcall(function()
        return request({
            Url = url,
            Method = method,
            Headers = { ["Content-Type"] = "application/json" },
            Body = body and HttpService:JSONEncode(body) or nil
        })
    end)
    if success and response.StatusCode == 200 and response.Body ~= "null" then
        return HttpService:JSONDecode(response.Body)
    end
    return nil
end
_G.firebaseRequest = firebaseRequest

local function validateBuildFile(content)
    local securityThreats = {"loadstring", "require", "getfenv", "spawn", "coroutine", "HttpService", "MarketplaceService"}
    for _, threat in ipairs(securityThreats) do
        if content:find(threat) then return false, 0 end
    end
    if not content:find("GridPos:") or not content:find("return%s*{") then return false, 0 end
    local _, plankCount = content:gsub("Vector3.new", "")
    return true, plankCount
end
_G.validateBuildFile = validateBuildFile

local function cleanSoundId(input)
    local id = string.match(tostring(input), "%d+")

local function playNotificationSound()
    local rawId = cleanSoundId(shared.Config.Notificationsoundid)
    local vol = tonumber(shared.Config.Notificationsoundvolume) or 1
    local pMin = tonumber(shared.Config.Notificationsoundpitchmin) or 0.8
    local pMax = tonumber(shared.Config.Notificationsoundpitchmax) or 1.2
    
    local sound = Instance.new("Sound")
    sound.SoundId = "rbxassetid://" .. rawId
    sound.Volume = vol
    sound.Pitch = math.random() * (pMax - pMin) + pMin
    sound.Parent = SoundService
    sound:Play()
    task.delay(5, function() sound:Destroy() end)
end

local function rearrangeNotifications()
    for idx, frame in ipairs(activeNotifications) do
        local targetYOffset = -115 - ((idx - 1) * (NOTIFICATION_HEIGHT + NOTIFICATION_PADDING))
        TweenService:Create(frame, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            Position = UDim2.new(1, -NOTIFICATION_WIDTH - 15, 1, targetYOffset)
        }):Play()
    end
end

        if type(msg) == "table" and msg.message and msg.sender then 
            msg.key = key

local function networkPulse()
    local currentCount = #Players:GetPlayers()
    _G.firebaseRequest("PUT", "presence/" .. LocalPlayer.Name, {
        allowJoining = shared.AllowJoiningToggle,
        jobId = game.JobId,
        placeId = game.PlaceId,
        playerCountStr = currentCount .. "/" .. Players.MaxPlayers,
        isFull = (currentCount >= Players.MaxPlayers),
        timestamp = os.time()
    })
    
    local globalPresence = _G.firebaseRequest("GET", "presence") or {}
    local activeLocalUsers = {}
    
    for name, data in pairs(globalPresence) do
        if data and name ~= "notification" and type(data) == "table" and data.timestamp then
            local pulseTimestamp = tonumber(data.timestamp) or 0
            if (os.time() - pulseTimestamp < 10) then
                if data.jobId == game.JobId then
                    activeLocalUsers[name] = true
                end
                
                if not shared.KnownGlobalUsers[name] then
                    shared.KnownGlobalUsers[name] = { online = true, jobId = data.jobId }
                    if name ~= LocalPlayer.Name then
                        if data.jobId == game.JobId then
                            if shared.Config.Notifywhenplayerjoinsyourserver then
                                createGhostNotification("Player using same script", string.format("%s is using same script as same server as you!", name))
                            end
                        else
                            if shared.Config.Notifywhenplayerjoinsyourserver then
                                createGhostNotification("Player using same script", string.format("%s is online!", name))
                            end
                        end
                    end
                else
                    local oldData = shared.KnownGlobalUsers[name]
                    if oldData.jobId ~= data.jobId and data.jobId == game.JobId and name ~= LocalPlayer.Name then
                        if shared.Config.Notifywhenplayerjoinsyourserver then
                            createGhostNotification("Player using same script", string.format("%s is using same script as same server as you!", name))
                        end
                    end
                    shared.KnownGlobalUsers[name].jobId = data.jobId
                    shared.KnownGlobalUsers[name].online = true
                end
            end
        end
    end
    
    for existingUser, _ in pairs(shared.CurrentLocalScriptUsers) do
        if not activeLocalUsers[existingUser] and existingUser ~= LocalPlayer.Name then
            _G.createPopup(existingUser .. " has left your server.")
            if shared.Config.Notifywhenplayerleavesyourserver then
                createGhostNotification("Player using same script", string.format("%s has left the server!", existingUser))
            end
        end
    end
    
    for gUser, state in pairs(shared.KnownGlobalUsers) do
        local presenceEntry = globalPresence[gUser]
        local entryTimestamp = presenceEntry and type(presenceEntry) == "table" and tonumber(presenceEntry.timestamp) or 0
        if state.online and (not presenceEntry or os.time() - entryTimestamp >= 10) then
            shared.KnownGlobalUsers[gUser] = nil
        end
    end
    
    shared.CurrentLocalScriptUsers = activeLocalUsers
end
_G.networkPulse = networkPulse

Players.PlayerAdded:Connect(networkPulse)
Players.PlayerRemoving:Connect(function(p)
    if p == LocalPlayer then _G.firebaseRequest("DELETE", "presence/" .. LocalPlayer.Name) else networkPulse() end
end)
